```bash
composer require somehow-digital/imgproxy
```

```php
use SomehowDigital\ImgProxy\Option\Height;
use SomehowDigital\ImgProxy\Option\Quality;
use SomehowDigital\ImgProxy\Option\Width;
use SomehowDigital\ImgProxy\Url;

$url = Url::create()
    ->source('https://example.com/image.jpg')
    ->options(new Width(300), new Height(200), new Quality(80));

$url->build('webp');
// _/w:300/h:200/q:80/plain/https%3A%2F%2Fexample.com%2Fimage.jpg@webp
```
